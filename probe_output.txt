(function () {
  let path = document.location.pathname;
  let programPath = '';
  let programName = '';
  const filters = {};
  const SEARCH_TXT = 'Search';
  const SEARCHING_TXT = 'Searching...';
  const LOADING_HTML = `<h3><i class="fa fa-spinner fa-spin"></i> Loading...</h3>`;
  let jurisdiction = null;
  let moreResults = true;
  let resultsTemplate = '';
  const urlparams = JSON.fromURL(document.location.search);
  let searchStr = urlparams.searchStr ?? '';
  let lat = urlparams.lat ?? 0;
  let lng = urlparams.lng ?? 0;
  let start = urlparams.start ?? 0;
  let count = urlparams.count ?? 20;
  const sort = {};

  // Captcha session storage
  function getCaptchaVerification() {
    const timestamp = sessionStorage.getItem('captchaInspectionsLandingPageVerified');
    if (!timestamp) return false;
    
    const now = Date.now();
    const sixtyMinutes = 60 * 60 * 1000;
    return (now - parseInt(timestamp)) < sixtyMinutes;
  }

  // Captcha verification for search
  function showCaptchaForSearch(callback) {
    const captchaOverlay = document.getElementById('captcha-overlay');
    const verifyButton = document.getElementById('verify-captcha');
    const captchaError = document.getElementById('captcha-error');

    // Show the overlay
    captchaOverlay.style.display = 'flex';

    // Reset captcha
    if (typeof grecaptcha !== 'undefined') {
      grecaptcha.reset();
    }

    // Handle verification
    const handleVerification = function() {
      const captchaResponse = grecaptcha.getResponse();
      
      if (!captchaResponse) {
        captchaError.style.display = 'block';
        captchaError.textContent = 'Please complete the captcha verification.';
        return;
      }

      // Store verification timestamp in session storage
      sessionStorage.setItem('captchaInspectionsLandingPageVerified', Date.now().toString());

      // Hide overlay
      captchaOverlay.style.display = 'none';
      captchaError.style.display = 'none';
      
      // Remove event listener
      verifyButton.removeEventListener('click', handleVerification);
      
      // Execute the callback (continue with search)
      callback();
    };

    verifyButton.addEventListener('click', handleVerification);
  }

  const getUserLocation = function () {
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        lat = pos.coords.latitude;
        lng = pos.coords.longitude;
		    start = 0;
        $('#loadingLocation').hide();
        $('#failedLocation').hide();
        $('#activeLocation').show();
        $('#useLocation').hide();
        $('.section---establishment-list').html('Refreshing...');
		    getData();
      },
      (err) => {
        console.error(`ERROR(${err.code}): ${err.message}`);
        $('#loadingLocation').hide();
        $('#failedLocation').show();
		    $('#activeLocation').hide();
      },
      {
        enableHighAccuracy: false,
        maximumAge: Infinity
      }
    );
  };

  const controller = async function () {
    resultsTemplate = $('#resultsTemplate').text();

    jurisdiction = await app.jurisdiction.getAsync();

    const fdEl = document.getElementById(`filterdate`);

    if (fdEl) {
      const minDateAttr = fdEl.getAttribute('data-minDate');
      const startDate = moment(minDateAttr || '01/01/' + new Date().getFullYear().toString());
      const endDate = moment();
      const minDate = minDateAttr ? moment(minDateAttr) : moment().subtract(10, 'year');

      // system specific date format, configured to be accepted by flatpickr
      const displayFormat = (jurisdiction.hscSettings?.dateFormat || "MM/DD/YYYY")
      .toUpperCase()
      .replace('MMM', 'M')
      .replace('MM', 'm')
      .replace('DD', 'd')
      .replace('YYYY', 'Y');
  
      fdEl.value = startDate.format('YYYY-MM-DD') + ' to ' + endDate.format('YYYY-MM-DD');
  
      const datePickerConfig = {
        enableTime: false,
        noCalendar: false,
        defaultDate: [startDate.toDate(), endDate.toDate()],
        minDate: minDate.toDate(),
        allowInput: true,
        altInput: true, 
        altFormat: displayFormat, // date format displayed to user... uses system date format
        altInputClass: `alt-datePicker`,
        appendTo: fdEl.parentNode,
        position() {},
        mode: 'range',
        onReady(a, b, fp) {
          fp.altInput.setAttribute('aria-label', 'Date Range');
        },
        onChange(a, b, fp) {
          // only run update once a start and enddate have been chosen
          if (a.length === 2) {
            start = 0;
            getData();
          }
        },
        onDayCreate(dObj, dStr, fp, dayElem) {
          if ($(dayElem).attr('tabindex') == -1) {
            $(dayElem).attr('tabindex', 0);
          }
        },
        onOpen(selectedDates, dateStr, instance) {
          instance.monthElements[0].setAttribute('tabindex', 0);
          instance.currentYearElement.setAttribute('tabindex', 0);
        },
      };
      // call flatpickr instance, attach it to the current field, pass in defined config options.
      const instance = flatpickr(fdEl, datePickerConfig);
      
      /*
        NOTE: Keeping this here as a reminder about removing these handy 'helper' buttons for specific ranges

        {
          ranges: {
            Today: [moment(), moment()],
            Yesterday: [moment().subtract(1, 'days'), moment().subtract(1, 'days')],
            'Last 7 Days': [moment().subtract(6, 'days'), moment()],
            'Last 30 Days': [moment().subtract(29, 'days'), moment()],
            'This Month': [moment().startOf('month'), moment().endOf('month')],
            'Last Month': [moment().subtract(1, 'month').startOf('month'), moment().subtract(1, 'month').endOf('month')],
          },
        }
      */
    }
    
    $('.filterItem').each(function (idx, itm) {
      if (itm.tagName === 'SELECT') {
        $(itm).on('change', () => {
          start = 0;
          getData();
        });
      }
    });

    $('.sortItem').each(function (idx, itm) {
      $(itm).on('change', () => {
        const selectedSort = $(itm).val().split('|');
        sort.field = selectedSort[0];
        sort.direction = selectedSort[1];

        start = 0;
        getData();
      });
    });

    $('.load-more-results-button').on('click', function () {
      if (this.innerHTML !== 'Loading...') {
        start += count;
        getData();
      }
    });

    $('.search-button').on('click', function () {
      doSearch(this);
    });

    $('#useLocation').on( 'click',function() {
      $('#loadingLocation').show();
      getUserLocation();
    });

    getData();
  };

  const doSearch = function(searchBtn) {
    // Check if captcha verification is required
    const captchaOverlay = document.getElementById('captcha-overlay');
    if (captchaOverlay && !getCaptchaVerification()) {
      // Show captcha and continue search after verification
      showCaptchaForSearch(() => doSearchInternal(searchBtn));
      return;
    }

    // Captcha not required or already verified, proceed with search
    doSearchInternal(searchBtn);
  }

  const doSearchInternal = function(searchBtn) {
    searchStr = $('#search').val()?.trim() || '';
    const hasSearchTemplate = jurisdiction?.programConstraints?.[programName]?.searchTemplate || jurisdiction?.searchTemplate ? true : false;
    const searchFirstOverride = jurisdiction?.displaySettings?.searchFirstOverride ?? false;

    if (hasSearchTemplate && !searchStr) {
      alert('You must enter search terms to do search');
      return;
    }

    // Indicate that we are searching for results
    $(searchBtn).html( SEARCHING_TXT );

    // Clear the current list of results
    $('.section---establishment-list').html(LOADING_HTML);

    // Hide the load more results section
    $('.load-more-results-button').hide();

    //if (hasSearchTemplate) {
    if (!searchFirstOverride) {
      let searchUrl = `/${path}${programName ? '/program-' + programName : ''}/search?searchStr=${searchStr}`;
      if (jurisdiction.searchUserLocation) {
        searchUrl += `&lat=${lat}&lng=${lng}`;
      }
      document.location = searchUrl;
      return;
    }

    getData();
  }

  const getData = async function() {
    // TODO: make configurable
    const viewEstTitle = "View Establishment";
    const viewInspTitle = "View Inspection";

    $('.filterItem').each(function (idx, itm) {
      const filterId = this.id.replace(/filter/, '');
      filters[filterId] = $(itm).val();
      if (filterId === 'date' && filters[filterId].length && !filters[filterId].includes(' to ')) {
        filters[filterId] = `${filters[filterId]} to ${filters[filterId]}`;
      }
    });
    if (start === 0) {
      $('.section---establishment-list').html(LOADING_HTML);
    }

    if (jurisdiction.inspectionSiteSearchFirst) {
      $('.load-more-results-button').html("Search above to find what you're looking for");
    }
    else {
      $('.load-more-results-button').html("Loading...");
    }

    let searchQueryOverride = null;
    const globalSearchQueryOverride = jurisdiction?.searchQueryOverride;
    const programSearchQueryOverride = jurisdiction?.programConstraints?.[programName]?.searchQueryOverride;

    searchQueryOverride = globalSearchQueryOverride || programSearchQueryOverride;

    try {
      const res = await app.ajax.requestAsync({
        data: {
          path,
          programName,
          filters,
          start,
          count,
          searchQueryOverride,
          searchStr,
          lat,
          lng,
          sort,
        },
        task: 'searchInspections',
      },
      {
        url: '/',
        contentType: 'application/json',
        responseType: 'json',
      });

      if (start === 0) {
        $('.section---establishment-list').html('');
      }

      let resultsHTML = `
        {if:inspections.length == 0}<h3>No results found</h3>{/if}
        {[inspections]}
          <div class="flex-row">
            <div style="flex-grow: 1;">
              <h4 class="establishment-list-name"><a href="/{path}{helpers.getProgramPath}/permit/?permitID={permitID}" title="${viewEstTitle}">{establishmentName}</a></h4>
              <div class="establishment-list-address">{addressLine1} {addressLine2} {city}, {state} {zip}</div>
              {if:distanceFromLatLng}
                {helpers.showDistance}
              {/if}
              {if:permitStatus}
                <div class="establishment-list-address">Status: {permitStatus}</div>
              {/if}
            </div>
            <div class="text-right">
              {if:helpers.showHeader}
                <div class="recent-inspection-section"><span class="text-span-3">MOST RECENT INSPECTION</span></div>
              {/if}
              <div class="text-right">{inspectionType} | {purpose}</div>
              <div class="text-right">
                {FORMATTERS.DATE_WRITTENUTC.inspectionDate}
                <span style="{if:jurisdiction.displaySettings.showScore == false}display:none;{/if}">
                  <span style="{if:helpers.hideScoreAndDisplay == true}display:none;{/if}">
                    | <strong>{if:helpers.hideScore == false}{score}{/if} {if:helpers.hideScoreDisplay == false}{scoreDisplay}{/if}</strong>
                  </span>
                </span>
              </div>
            </div>
            {helpers.renderDetailLink}
          </div>
        {[/inspections]}`;

      if (resultsTemplate.trim().length > 1) {
        resultsHTML = resultsTemplate;
      }

      if (!res.err) {
        $('.section---establishment-list').html(
          $('.section---establishment-list').html() +
            resultsHTML.compile({
              inspections: res,
              path: path,
              jurisdiction: jurisdiction,
              firstItem: (res[0] || {}).inspectionID,
              start: start,
              count: count,
              helpers: {
                showHeader: function (row) {
                  return start === 0 && row.__i === 0;
                },
                showDistance: function(row) {
                  let out = '';
                  if( row.distanceFromLatLng ) {
                    // convert km to miles
                    out = parseInt(row.distanceFromLatLng / 1609).toLocaleString();
                    out = `<div><i>approx ${out} mi away</i></div>`;
                  }
                  return out;
                },
                getProgramPath: function () {
                  return programPath ? '/' + programPath : '';
                },
                renderDetailLink: function (row) {
                  
                  if (
                    jurisdiction.displaySettings &&
                    jurisdiction.displaySettings.disableDetailView &&
                    jurisdiction.displaySettings.allowPDFPrint
                  ) {
                    return `<div class="establishment-list-button-column"><a href="/${path}${programPath ? '/' + programPath : ''}/print/?task=getPrintable&path=${path}&pKey=${row.inspectionID}" target="_blank" class="view-inspections-button w-button">${viewInspTitle} <span class="sr-only">(opens in a new tab)</span></a></div>`;
                  } else {
                    return `<div class="establishment-list-button-column"><a href="/${path}${programPath ? '/' + programPath : ''}/inspection/?inspectionID=${row.inspectionID}" class="view-inspections-button w-button">${viewInspTitle}</a></div>`;
                  }
                },
                hideScore: function (row) {
                  let inspType = row.inspectionType ? row.inspectionType : '';
                  let purpose = row.purpose ? row.purpose : '';

                  if (!jurisdiction.displaySettings || !jurisdiction.displaySettings.scoreDisplaySettings) return false;

                  if (!jurisdiction.displaySettings.scoreDisplaySettings[inspType]) return false;

                  if (
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes &&
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes.includes(purpose)
                  )
                    return true;

                  return !jurisdiction.displaySettings.scoreDisplaySettings[inspType].showScore;
                },
                hideScoreDisplay: function (row) {
                  let inspType = row.inspectionType ? row.inspectionType : '';
                  let purpose = row.purpose ? row.purpose : '';

                  if (!jurisdiction.displaySettings || !jurisdiction.displaySettings.scoreDisplaySettings) return false;

                  if (!jurisdiction.displaySettings.scoreDisplaySettings[inspType]) return false;

                  if (
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes &&
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes.includes(purpose)
                  )
                    return true;

                  return !jurisdiction.displaySettings.scoreDisplaySettings[inspType].showScoreDisplay;
                },
                hideScoreAndDisplay: function (row) {
                  let inspType = row.inspectionType ? row.inspectionType : '';
                  let purpose = row.purpose ? row.purpose : '';

                  if (!jurisdiction.displaySettings || !jurisdiction.displaySettings.scoreDisplaySettings) return false;

                  if (!jurisdiction.displaySettings.scoreDisplaySettings[inspType]) return false;

                  if (
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes &&
                    jurisdiction.displaySettings.scoreDisplaySettings[inspType].noScorePurposes.includes(purpose)
                  )
                    return true;

                  return (
                    !jurisdiction.displaySettings.scoreDisplaySettings[inspType].showScoreDisplay &&
                    !jurisdiction.displaySettings.scoreDisplaySettings[inspType].showScore
                  );
                },
                hideLegacy: function (row) {
                  return row.inspectionType.replace('(OLD)', '');
                },
              },
            })
        );
      }

      if (res.length < count) {
        moreResults = false;
        $('.load-more-results-button').hide();
      } else {
        moreResults = true;
        $('.load-more-results-button').show();
      }
    }
    catch(e) {
      $('.section---establishment-list').html('<h3>Error loading results</h3>');
    }
    
    if (!jurisdiction.inspectionSiteSearchFirst) {
      $('.load-more-results-button').html('Load More Results');
    }

    $('.search-button').html(SEARCH_TXT);
  }

  programPath = document.location.pathname.split('/')[2] || '';
  programName = '';

  if (programPath && programPath.left(7) === 'program') {
    programName = programPath.replace('program-', '');
  }

  if (path.substr(0, 1) === '/') {
    path = path.substr(1);
    path = path.split('/')[0];
    path = path.split('?')[0];
  }

  app.onInit(controller);
})();
